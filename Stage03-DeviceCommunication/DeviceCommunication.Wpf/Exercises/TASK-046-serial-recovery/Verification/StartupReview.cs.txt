using System;
using System.IO;
using System.Text.Json;
using System.Windows;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using System.Windows.Threading;
using DeviceCommunication.Wpf.Exercises.Task046;

public static class StartupReview
{
    [STAThread]
    public static void Main(string[] args)
    {
        Application app = new Application();
        app.ShutdownMode = ShutdownMode.OnExplicitShutdown;
        Task046Window window = new Task046Window();
        window.Left = -20000;
        window.Top = -20000;
        window.Width = window.MinWidth;
        window.Height = window.MinHeight;
        window.Show();
        window.UpdateLayout();
        window.Dispatcher.Invoke(DispatcherPriority.ApplicationIdle, new Action(() => { }));
        bool loaded = window.IsLoaded;
        RenderTargetBitmap bitmap = new RenderTargetBitmap((int)window.ActualWidth, (int)window.ActualHeight, 96, 96, PixelFormats.Pbgra32);
        bitmap.Render(window);
        PngBitmapEncoder encoder = new PngBitmapEncoder();
        encoder.Frames.Add(BitmapFrame.Create(bitmap));
        using (FileStream output = File.Create(Path.Combine(args[0], "startup-minimum-2026-10-19.png")))
        {
            encoder.Save(output);
        }
        window.Close();
        window.Dispatcher.Invoke(DispatcherPriority.ApplicationIdle, new Action(() => { }));
        app.Shutdown();
        Console.WriteLine(JsonSerializer.Serialize(new { loaded, closed = !window.IsVisible, serialPortOpened = false, learnerTodosVerified = false }));
    }
}
