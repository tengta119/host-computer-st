using TcpUdp.Wpf.Exercises.Task050;

public static class StartupReview
{
    private static readonly List<string> Checks = new List<string>();
    private static void Check(string name, bool condition)
    {
        if (!condition) throw new Exception(name);
        Checks.Add(name);
    }
    private static T Field<T>(object target, string name)
    {
        return (T)target.GetType().GetField(name, BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(target)!;
    }
    private static object? Call(object target, string name, params object[] args)
    {
        return target.GetType().GetMethod(name, BindingFlags.Instance | BindingFlags.NonPublic)!.Invoke(target, args);
    }
    private static void Click(Task051Window window, string name)
    {
        ((Button)window.FindName(name)).RaiseEvent(new RoutedEventArgs(Button.ClickEvent));
    }
    private static int FreePort()
    {
        TcpListener listener = new TcpListener(IPAddress.Loopback, 0);
        listener.Start();
        int port = ((IPEndPoint)listener.LocalEndpoint).Port;
        listener.Stop();
        return port;
    }
    private static bool InsideContent(Task051Window window, string name)
    {
        FrameworkElement child = (FrameworkElement)window.FindName(name);
        FrameworkElement content = (FrameworkElement)window.Content;
        Rect rect = child.TransformToAncestor(content).TransformBounds(new Rect(0, 0, child.ActualWidth, child.ActualHeight));
        return child.ActualWidth > 0 && child.ActualHeight > 0 && rect.Left >= -1 && rect.Top >= -1 && rect.Right <= content.ActualWidth + 1 && rect.Bottom <= content.ActualHeight + 1;
    }

    private static async Task VerifyFaultMode(string mode, string command, string expected)
    {
        int port = FreePort();
        using (CancellationTokenSource stop = new CancellationTokenSource())
        {
            Task051Simulator simulator = new Task051Simulator();
            Task server = simulator.RunAsync(port, mode, stop.Token);
            Task051TcpConnection connection = new Task051TcpConnection();
            try
            {
                await connection.ConnectAsync("127.0.0.1", port, CancellationToken.None);
                Check(mode + "仍可查询设备状态", await connection.RequestAsync("GET_STATUS", CancellationToken.None) == "STOPPED");
                string actual = "未发生预期故障";
                try { await connection.RequestAsync(command, CancellationToken.None); }
                catch (Task051TimeoutException exception) { actual = exception.Operation; }
                catch (IOException) { actual = "对端关闭"; }
                Check(mode + "返回可区分的故障", actual == expected);
                connection.Close();
                await connection.ConnectAsync("127.0.0.1", port, CancellationToken.None);
                Check(mode + "旧迟到回复不进入新状态查询", await connection.RequestAsync("GET_STATUS", CancellationToken.None) == "STOPPED");
                if (mode == "silent-temp")
                {
                    using (CancellationTokenSource cancel = new CancellationTokenSource())
                    {
                        cancel.CancelAfter(50);
                        bool userCancellation = false;
                        try { await connection.RequestAsync("GET_TEMP", cancel.Token); }
                        catch (OperationCanceledException) { userCancellation = true; }
                        Check("外部取消不误记为超时", userCancellation);
                    }
                }
            }
            finally
            {
                connection.Close();
                stop.Cancel();
                await server;
            }
        }
    }

    [STAThread]
    public static void Main(string[] args)
    {
        Application app = new Application();
        app.ShutdownMode = ShutdownMode.OnExplicitShutdown;
        Task051Window window = new Task051Window();
        window.Left = -20000;
        window.Top = -20000;
        window.Show();
        app.Dispatcher.BeginInvoke(new Action(async () =>
        {
            string? failure = null;
            try
            {
                window.UpdateLayout();
                Check("真实窗口已加载", window.IsLoaded);
                Check("初始未连接与操作禁用", ((Button)window.FindName("ConnectButton")).IsEnabled && !((Button)window.FindName("ReadButton")).IsEnabled);
                Check("初始自动重连和心跳均未启动", !Field<DispatcherTimer>(window, "_reconnectTimer").IsEnabled && !Field<DispatcherTimer>(window, "_heartbeatTimer").IsEnabled);
                Check("心跳判断TODO仍未实现", !(bool)Call(window, "CanSendHeartbeat")!);

                // 直接注入布局样例，不调用学员ApplyTemperature，不把显示验证当作学员实现。
                ObservableCollection<Task050Reading> history = Field<ObservableCollection<Task050Reading>>(window, "_history");
                history.Add(new Task050Reading { ReceivedAt = DateTime.Now, Temperature = 35.6 });
                window.UpdateLayout();
                Check("显示支持可呈现单点温度与报警", ((TextBlock)window.FindName("LatestText")).Text.Contains("35.6") && ((TextBlock)window.FindName("AlarmText")).Text == "温度超过阈值，报警。");
                history.Add(new Task050Reading { ReceivedAt = DateTime.Now, Temperature = 34.8 });
                history.Add(new Task050Reading { ReceivedAt = DateTime.Now, Temperature = 36.2 });
                window.UpdateLayout();
                Check("历史表格绑定同一集合", Object.ReferenceEquals(((DataGrid)window.FindName("HistoryGrid")).ItemsSource, history) && history.Count == 3);
                ((TextBox)window.FindName("ThresholdInput")).Text = "40";
                Call(window, "ApplyThresholdClick", window, new RoutedEventArgs());
                Check("阈值支持重新判断已有温度", ((TextBlock)window.FindName("AlarmText")).Text == "温度未超过阈值。");
                ((TextBox)window.FindName("ThresholdInput")).Text = "NaN";
                Call(window, "ApplyThresholdClick", window, new RoutedEventArgs());
                Check("非法阈值被拒绝", ((TextBlock)window.FindName("OperationText")).Text.Contains("0至100"));
                ((TextBox)window.FindName("ThresholdInput")).Text = "35";
                Call(window, "ApplyThresholdClick", window, new RoutedEventArgs());
                for (int i = 0; i < 205; i++) Call(window, "RecordLog", "布局样例", "仅验证显示：127.0.0.1:5000，收到完整温度35.6，具体来源与操作保留。");
                Check("教练日志支持限制200条", Field<ObservableCollection<Task050LogEntry>>(window, "_logs").Count == 200);
                string[] controls = { "HostInput", "PortInput", "ConnectButton", "DisconnectButton", "ReadButton", "StartDeviceButton", "StopDeviceButton", "StatusButton", "StartCollectionButton", "StopCollectionButton", "ClearButton", "HistoryGrid", "ChartCanvas", "LogGrid", "OperationText", "AutoReconnectCheckBox", "RecoveryText", "HeartbeatText", "HeartbeatButton" };
                foreach (bool minimum in new bool[] { false, true })
                {
                    window.Width = minimum ? window.MinWidth : 1180;
                    window.Height = minimum ? window.MinHeight : 900;
                    window.UpdateLayout();
                    await window.Dispatcher.InvokeAsync(new Action(() => { }), DispatcherPriority.ApplicationIdle);
                    bool visible = true;
                    foreach (string name in controls) visible = visible && InsideContent(window, name);
                    Check(minimum ? "最小尺寸控件均在客户区内" : "默认尺寸控件均在客户区内", visible);
                    Check(minimum ? "最小尺寸表格日志和曲线有可用高度" : "默认尺寸表格日志和曲线有可用高度", ((DataGrid)window.FindName("HistoryGrid")).ActualHeight > 140 && ((DataGrid)window.FindName("LogGrid")).ActualHeight > 90 && ((Canvas)window.FindName("ChartCanvas")).ActualHeight >= 125);
                }
                RenderTargetBitmap bitmap = new RenderTargetBitmap((int)window.ActualWidth, (int)window.ActualHeight, 96, 96, PixelFormats.Pbgra32);
                bitmap.Render(window);
                PngBitmapEncoder encoder = new PngBitmapEncoder();
                encoder.Frames.Add(BitmapFrame.Create(bitmap));
                using (FileStream output = File.Create(Path.Combine(args[0], "startup-minimum-2026-10-20.png"))) encoder.Save(output);
                Click(window, "ClearButton");
                Check("清空支持恢复空历史", history.Count == 0 && ((TextBlock)window.FindName("LatestText")).Text.Contains("—"));

                int port = FreePort();
                using (CancellationTokenSource simulatorStop = new CancellationTokenSource())
                {
                    Task051Simulator simulator = new Task051Simulator();
                    Task server = simulator.RunAsync(port, "normal", simulatorStop.Token);
                    Task051TcpConnection connection = new Task051TcpConnection();
                    try
                    {
                        await connection.ConnectAsync("127.0.0.1", port, CancellationToken.None);
                        Check("无菜单连接辅助实际连接模拟器", connection.IsConnected);
                        Check("无菜单辅助取得完整温度", await connection.RequestAsync("GET_TEMP", CancellationToken.None) == "35.6");
                        Check("无菜单辅助取得启动确认", await connection.RequestAsync("START_DEVICE", CancellationToken.None) == "OK");
                        Check("无菜单辅助查询真实运行状态", await connection.RequestAsync("GET_STATUS", CancellationToken.None) == "RUNNING");
                        connection.Close();
                        await connection.ConnectAsync("127.0.0.1", port, CancellationToken.None);
                        Check("无菜单辅助关闭后可手动重连", await connection.RequestAsync("GET_STATUS", CancellationToken.None) == "RUNNING");
                        Check("无菜单辅助停止设备", await connection.RequestAsync("STOP_DEVICE", CancellationToken.None) == "OK" && await connection.RequestAsync("GET_STATUS", CancellationToken.None) == "STOPPED");
                    }
                    finally
                    {
                        connection.Close();
                        simulatorStop.Cancel();
                        await server;
                    }
                }

                await VerifyFaultMode("silent-temp", "GET_TEMP", "业务请求");
                await VerifyFaultMode("delay-temp", "GET_TEMP", "业务请求");
                await VerifyFaultMode("silent-ping", "PING", "心跳");
                await VerifyFaultMode("close-temp", "GET_TEMP", "对端关闭");
                // 直接注入恢复等待，仅验证教练提供的取消和上限保护，不实现TODO 2的调度。
                typeof(Task051Window).GetField("_wantConnected", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(window, true);
                ((CheckBox)window.FindName("AutoReconnectCheckBox")).IsChecked = true;
                Field<DispatcherTimer>(window, "_reconnectTimer").Start();
                ((CheckBox)window.FindName("AutoReconnectCheckBox")).IsChecked = false;
                Check("取消自动选项停止教练恢复定时器", !Field<DispatcherTimer>(window, "_reconnectTimer").IsEnabled);
                ((CheckBox)window.FindName("AutoReconnectCheckBox")).IsChecked = true;
                typeof(Task051Window).GetField("_reconnectAttempts", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(window, 5);
                Call(window, "ScheduleReconnect");
                Check("五次上限保护不再安排连接", !Field<DispatcherTimer>(window, "_reconnectTimer").IsEnabled && !Field<bool>(window, "_wantConnected"));
                typeof(Task051Window).GetField("_wantConnected", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(window, true);
                Field<DispatcherTimer>(window, "_reconnectTimer").Start();
                Click(window, "DisconnectButton");
                Check("主动断开停止全部恢复意图", !Field<bool>(window, "_wantConnected") && !Field<DispatcherTimer>(window, "_reconnectTimer").IsEnabled);

                TcpListener silentPeer = new TcpListener(IPAddress.Loopback, 0);
                silentPeer.Start();
                try
                {
                    int silentPort = ((IPEndPoint)silentPeer.LocalEndpoint).Port;
                    Task<TcpClient> accepting = silentPeer.AcceptTcpClientAsync();
                    Task051TcpConnection connection = Field<Task051TcpConnection>(window, "_connection");
                    CancellationTokenSource session = new CancellationTokenSource();
                    typeof(Task051Window).GetField("_sessionSource", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(window, session);
                    await connection.ConnectAsync("127.0.0.1", silentPort, session.Token);
                    using (TcpClient peer = await accepting)
                    {
                        CancellationToken token = session.Token;
                        Task<string> waiting = connection.RequestAsync("GET_TEMP", token);
                        window.Close();
                        bool stopped = false;
                        try { await waiting; }
                        catch (OperationCanceledException) { stopped = true; }
                        catch (IOException) { stopped = true; }
                        catch (ObjectDisposedException) { stopped = true; }
                        Check("关闭窗口取消教练连接辅助的未完成接收", stopped && token.IsCancellationRequested && !connection.IsConnected);
                        Check("关闭窗口停止全部定时器", !Field<DispatcherTimer>(window, "_pollingTimer").IsEnabled && !Field<DispatcherTimer>(window, "_heartbeatTimer").IsEnabled && !Field<DispatcherTimer>(window, "_reconnectTimer").IsEnabled);
                        Check("关闭后旧连接结果被拒绝", !(bool)Call(window, "IsCurrentSession", Field<int>(window, "_sessionId"))!);
                    }
                }
                finally
                {
                    silentPeer.Stop();
                }
                Check("窗口已实际关闭", !window.IsVisible);
            }
            catch (Exception exception)
            {
                failure = exception.ToString();
                if (window.IsVisible) window.Close();
            }
            Console.WriteLine("PREPARATION_RESULT=" + JsonSerializer.Serialize(new { passed = Checks.Count, checks = Checks, failure, injectedHistoryRows = 3, learnerTodosVerified = false }));
            app.Shutdown(failure == null ? 0 : 1);
        }));
        app.Run();
    }
}
