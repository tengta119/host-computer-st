using TcpUdp.Exercises.Task052;
using System.Net;
using System.Net.Sockets;

if (args[0] == "server")
{
    using CancellationTokenSource stop = new CancellationTokenSource();
    Task server = new Task052Simulator().RunAsync(Int32.Parse(args[1]), args[2], stop.Token);
    Console.ReadLine();
    stop.Cancel();
    await server;
    Console.WriteLine("SERVER_STOPPED_CLEANLY");
}
else
{
    List<string> checks = new List<string>();
    async Task CheckStub(string name, Func<Task> action)
    {
        try { await action(); throw new Exception("Expected stub: " + name); }
        catch (NotImplementedException) { checks.Add(name); }
    }
    Task052SocketExercise tcp = new Task052SocketExercise();
    using Socket socket = new Socket(AddressFamily.InterNetwork, SocketType.Stream, ProtocolType.Tcp);
    await CheckStub("connect_stub", () => tcp.ConnectCoreAsync(socket, new IPEndPoint(IPAddress.Loopback, 1), CancellationToken.None));
    await CheckStub("send_stub", () => tcp.SendAllAsync(socket, new byte[4], CancellationToken.None));
    await CheckStub("receive_stub", () => tcp.ReceiveCoreAsync(socket, new byte[3], CancellationToken.None));
    using Task052UdpExercise udp = new Task052UdpExercise();
    using UdpClient raw = new UdpClient();
    await CheckStub("udp_send_stub", () => udp.SendQueryAsync(raw, new byte[3], new IPEndPoint(IPAddress.Loopback, 1), CancellationToken.None));
    UdpReceiveResult datagram = new UdpReceiveResult(System.Text.Encoding.ASCII.GetBytes("R|id|GET_TEMP|35.6"), new IPEndPoint(IPAddress.Loopback, 1));
    if (udp.TryAcceptDatagram(datagram, datagram.RemoteEndPoint, "id", "GET_TEMP", out _)) throw new Exception("Filter stub unexpectedly accepts.");
    checks.Add("filter_stub_rejects");
    using CancellationTokenSource cancelled = new CancellationTokenSource();
    cancelled.Cancel();
    try { await udp.QueryAsync(1, "GET_TEMP", cancelled.Token); throw new Exception("Expected cancellation."); }
    catch (OperationCanceledException) { checks.Add("external_cancel_before_send"); }
    Console.WriteLine(System.Text.Json.JsonSerializer.Serialize(checks));
}
