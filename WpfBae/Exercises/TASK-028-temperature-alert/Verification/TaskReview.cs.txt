using System.Reflection;
using System.Collections.ObjectModel;
using System.IO;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using System.Windows.Threading;
using WpfBae.Exercises.Task028;

internal static class TaskReview
{
    [STAThread]
    private static int Main(string[] args)
    {
        string output = Path.GetFullPath(args[0]);
        var checks = new List<object>();
        bool passed = true;
        Application app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        Task028Window? window = null;
        try
        {
            window = new Task028Window();
            window.Show();
            Pump(250);
            Check("实际窗口显示", window.IsLoaded && window.IsVisible);
            var timer = (DispatcherTimer)typeof(Task028Window).GetField("_samplingTimer", BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(window)!;
            var history = (ObservableCollection<double>)typeof(Task028Window).GetField("_temperatures", BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(window)!;
            var table = (DataGrid)window.FindName("HistoryDataGrid");
            Check("初始停止与单一原集合", !timer.IsEnabled && history.Count == 0 && ReferenceEquals(table.ItemsSource, history));
            Check("间隔一秒", timer.Interval == TimeSpan.FromSeconds(1));
            Check("初始提示占位", ((TextBlock)window.FindName("AlarmText")).Text == "暂无温度");
            string Alarm() { return ((TextBlock)window.FindName("AlarmText")).Text; }
            void State(string name, string expected)
            {
                Pump(30);
                Check(name + "/提示", Alarm() == expected);
                Check(name + "/原集合表格与计数", ReferenceEquals(table.ItemsSource, history) && table.Items.Count == history.Count && ((TextBlock)window.FindName("ReadingCountText")).Text == "完整历史条数：" + history.Count);
                var points = ((System.Windows.Shapes.Polyline)window.FindName("TemperatureLine")).Points;
                int count = Math.Min(3, history.Count);
                Check(name + "/最新三点", points.Count == count && points.Select((point, slot) => Math.Abs(point.X - new double[] {100,280,460}[slot]) < 0.01 && Math.Abs(point.Y - (240 - (history[history.Count-count+slot]-20)*20)) < 0.01).All(value=>value));
            }
            double[] samples = {24, 26.5, 25, 28, 24};
            for(int i=0;i<samples.Length;i++)
            {
                Click("AddButton");
                Check("手动第"+(i+1)+"条原温度", history.Count==i+1 && history[i]==samples[i]);
                State("手动"+samples[i], samples[i]>=27 ? "温度超限" : "温度正常");
                if(i==3) Snapshot("over-limit.png");
            }
            Click("ClearButton");
            State("正常后清空", "暂无温度");
            Check("清空原历史与图", history.Count==0 && ((FrameworkElement)window.FindName("SinglePointMarker")).Visibility==Visibility.Collapsed);
            Click("ClearButton");
            State("重复空集合清空", "暂无温度");
            // 直接向同一集合加入27与26.99，验证边界且排除从固定样例数组或图中槽位判断。
            history.Add(27);
            State("恰好27", "温度超限");
            history.Add(26.99);
            State("低于27", "温度正常");
            Click("ClearButton");
            Click("StartButton");
            Click("StartButton");
            Check("重复启动与采集中禁止清空", timer.IsEnabled && !((Button)window.FindName("ClearButton")).IsEnabled);
            // 调用实际处理器验证运行保护，防止仅靠按钮禁用掩盖方法问题。
            typeof(Task028Window).GetMethod("ClearButton_Click", BindingFlags.Instance|BindingFlags.NonPublic)!.Invoke(window, new object[]{window, new RoutedEventArgs()});
            for(int i=0;i<5;i++)
            {
                var watch=System.Diagnostics.Stopwatch.StartNew();
                while(history.Count < i+1 && watch.ElapsedMilliseconds < 1800) Pump(20);
                Check("自动第"+(i+1)+"条", history.Count==i+1 && history[i]==samples[i]);
                State("自动"+samples[i], samples[i]>=27 ? "温度超限":"温度正常");
            }
            Click("StopButton");
            Click("StopButton");
            int stoppedCount=history.Count;
            string stoppedAlarm=Alarm();
            Pump(2200);
            Check("停止2200ms保留历史与提示", !timer.IsEnabled && history.Count==stoppedCount && Alarm()==stoppedAlarm && ((Button)window.FindName("ClearButton")).IsEnabled);
            Click("StartButton");
            var resumeWatch=System.Diagnostics.Stopwatch.StartNew();
            while(history.Count==stoppedCount && resumeWatch.ElapsedMilliseconds<1800) Pump(20);
            Click("StopButton");
            Check("停止继续保留样例顺序", history.Count==stoppedCount+1 && history.Last()==26.5);
            State("继续后提示", "温度正常");
            Click("ClearButton");
            State("自动停止后清空", "暂无温度");
            Click("StartButton");
            var restartWatch=System.Diagnostics.Stopwatch.StartNew();
            while(history.Count==0 && restartWatch.ElapsedMilliseconds<1800) Pump(20);
            Click("StopButton");
            Check("清空重启从24", history.Count==1 && history[0]==24);
            State("清空重启提示", "温度正常");
            for(int i=0;i<3;i++) Click("AddButton");
            State("超限后停止保留", "温度超限");
            Click("StopButton");
            State("再次停止超限保留", "温度超限");
            Click("ClearButton");
            State("超限后清空", "暂无温度");
            Check("表格只读与隐藏临时按钮", table.IsReadOnly && !table.CanUserAddRows && !table.CanUserDeleteRows && !table.CanUserSortColumns && ((Button)window.FindName("ObserveButton")).Visibility==Visibility.Collapsed);
            Click("AddButton");
            CheckLayout("默认布局");
            Snapshot("default.png");
            window.Width = window.MinWidth;
            window.Height = window.MinHeight;
            Pump(250);
            CheckLayout("最小布局");
            Snapshot("minimum.png");
            Click("ClearButton");
            Check("停止状态清空", history.Count == 0);
            Click("StartButton");
            window.Close();
            Check("关闭收尾停止定时器", !timer.IsEnabled && !window.IsVisible);
            app.Shutdown();

            void Click(string name)
            {
                ((Button)window.FindName(name)).RaiseEvent(new RoutedEventArgs(Button.ClickEvent));
            }
            void CheckLayout(string prefix)
            {
                FrameworkElement root = (FrameworkElement)window.Content;
                foreach (string name in new[] { "StartButton", "StopButton", "AlarmText", "ClearButton", "SamplingStateText", "TickCountText", "ReadingCountText", "ChartCanvas", "HistoryDataGrid", "FirstReadingLabel" })
                {
                    var element = (FrameworkElement)window.FindName(name);
                    Point p = element.TranslatePoint(new Point(0, 0), root);
                    Check(prefix + "/" + name, element.ActualWidth > 0 && element.ActualHeight > 0 && p.X >= -1 && p.Y >= -1 && p.X + element.ActualWidth <= root.ActualWidth + 1 && p.Y + element.ActualHeight <= root.ActualHeight + 1);
                }
                var canvas = (FrameworkElement)window.FindName("ChartCanvas");
                var footer = (FrameworkElement)((System.Windows.Controls.Grid)root).Children[2];
                Point bottom = canvas.TranslatePoint(new Point(0, canvas.ActualHeight), root);
                Point top = footer.TranslatePoint(new Point(0, 0), root);
                Check(prefix + "/图表不遮挡底部说明", bottom.Y <= top.Y);
            }
            void Snapshot(string name)
            {
                FrameworkElement root = (FrameworkElement)window.Content;
                var bitmap = new RenderTargetBitmap((int)Math.Ceiling(root.ActualWidth), (int)Math.Ceiling(root.ActualHeight), 96, 96, PixelFormats.Pbgra32);
                var backdrop = new DrawingVisual();
                using (var drawing = backdrop.RenderOpen())
                {
                    drawing.DrawRectangle(Brushes.White, null, new Rect(0, 0, root.ActualWidth, root.ActualHeight));
                }
                bitmap.Render(backdrop);
                bitmap.Render(root);
                var encoder = new PngBitmapEncoder();
                encoder.Frames.Add(BitmapFrame.Create(bitmap));
                using var stream = File.Create(Path.Combine(output, name));
                encoder.Save(stream);
            }
        }
        catch (Exception error)
        {
            passed = false;
            checks.Add(new { name = "运行异常", passed = false, error = error.ToString() });
            window?.Close();
        }
        File.WriteAllText(Path.Combine(output, "results.json"), JsonSerializer.Serialize(new { purpose = "TASK-028实际窗口DoD审查", passed, checks }, new JsonSerializerOptions { WriteIndented = true }));
        return passed ? 0 : 1;

        void Check(string name, bool result)
        {
            passed &= result;
            checks.Add(new { name, passed = result });
        }
    }
    private static void Pump(int milliseconds)
    {
        var frame = new DispatcherFrame();
        var stop = new DispatcherTimer { Interval = TimeSpan.FromMilliseconds(milliseconds) };
        stop.Tick += (sender, e) => { stop.Stop(); frame.Continue = false; };
        stop.Start();
        Dispatcher.PushFrame(frame);
    }
}
