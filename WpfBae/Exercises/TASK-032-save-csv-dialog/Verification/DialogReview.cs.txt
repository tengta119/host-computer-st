            void Export(string? path,bool? overwrite=null)
            {
                bool seen=false,confirmation=false;string? error=null;
                var worker=Task.Run(()=>
                {
                    try
                    {
                        IntPtr main=IntPtr.Zero;
                        for(int i=0;i<80;i++){main=Dialogs().FirstOrDefault(h=>Title(h)=="选择温度历史CSV保存位置");if(main!=IntPtr.Zero)break;Thread.Sleep(100);}
                        if(main==IntPtr.Zero)throw new Exception("保存窗口未出现");seen=true;Thread.Sleep(200);
                        if(path==null){PostMessage(main,0x111,new IntPtr(2),IntPtr.Zero);return;}
                        IntPtr edit=IntPtr.Zero;EnumChildWindows(main,(h,l)=>{if(Class(h)=="Edit" && GetDlgCtrlID(h)==1001)edit=h;return true;},IntPtr.Zero);
                        if(edit==IntPtr.Zero)throw new Exception("文件名输入框未找到");SendMessage(edit,0x000C,IntPtr.Zero,path);PostMessage(main,0x111,new IntPtr(1),IntPtr.Zero);
                        for(int i=0;i<80;i++)
                        {
                            Thread.Sleep(100);var active=Dialogs();if(!active.Contains(main))return;
                            var other=active.FirstOrDefault(h=>h!=main);
                            if(other==IntPtr.Zero)continue;confirmation=true;
                            if(overwrite==true)PostMessage(other,0x111,new IntPtr(6),IntPtr.Zero);
                            else{PostMessage(other,0x111,new IntPtr(7),IntPtr.Zero);Thread.Sleep(200);PostMessage(main,0x111,new IntPtr(2),IntPtr.Zero);}return;
                        }
                        throw new Exception("保存窗口操作超时");
                    }
                    catch(Exception ex){error=ex.Message;foreach(var d in Dialogs())PostMessage(d,0x10,IntPtr.Zero,IntPtr.Zero);}
                });
                Click("ExportHistoryButton");worker.GetAwaiter().GetResult();Pump(50);
                Check("实际保存窗口/"+(path??"取消"),seen && error==null);
                if(overwrite.HasValue)Check("已有文件出现覆盖确认/"+path,confirmation);
                if(error!=null)throw new Exception(error);
            }

    static void Pump(int ms){var frame=new DispatcherFrame();var timer=new DispatcherTimer{Interval=TimeSpan.FromMilliseconds(ms)};timer.Tick+=(s,e)=>{timer.Stop();frame.Continue=false;};timer.Start();Dispatcher.PushFrame(frame);}
